\section{Pipeline}

\textbf{Policy:} MLP, 3$\times$512 ReLU, run every 0.1\,s.

\textbf{Inputs} (84, perfect state, fixed scaling):
\begin{itemize}
  \item Joint positions (7) and commanded joint velocities (7).
  \item Gripper: commanded and measured opening (2); pose (9: position and two rotation columns).
  \item Grasp-pose error in the gripper frame (9: position, coarse and fine, and rotation).
  \item Object: velocity (3) and width (1).
  \item 4 nearest obstacles, 11 each (44): position relative to the gripper, velocity, half extents, distance to the
    robot, present flag.
  \item Timing (2): time until the object enters and leaves reach.
\end{itemize}
\textbf{Outputs:} 8 means $\mu$. The noise $\sigma$ is a learned, state-independent parameter per dimension (not a
network output); at test time $u=\mu$.
\begin{itemize}
  \item Joints (7): $u_{1:7}\sim\mathcal N(\mu_{1:7},\sigma^2)$, $a=\tanh u$; $a\,v_{\max}$ is the target joint velocity,
    reached by a ramp clipped at 0.6$\times$ the reference accelerations, sent as positions every 10\,ms.
  \item Gripper (1): $u_8\sim\mathcal N(\tanh\mu_8,\sigma_8^2)$ in action units, clipped to $[-1,1]$, mapped linearly to
    the width ($-1$ closed, $+1$ open).
\end{itemize}
\textbf{Scenes:} one training mix, no curriculum: 60\% like the held-out generator (0--4 obstacles, a quarter
free-flying), 40\% blockers (post, plank, pillar); fresh seeds, never one of the 214 test scenes.

\textbf{1. BC} on 34{,}981 successful (C) episodes (4.4M pairs; scenes from no obstacles up to the training mix, no compute latency), $s'=s+0.02\,\epsilon$:
\[
  L_{\text{BC}}=\tfrac18\textstyle\sum_i\big(\tanh\mu_i(s')-a^*_i\big)^2
  +\tfrac18\sum_i\Big[\tfrac{(\operatorname{atanh}a^*_i-\sg\mu_i)^2}{2\sigma_i(s)^2}+\log\sigma_i(s)\Big].
\]
Adam $3\cdot10^{-4}$, batch 512, 40 epochs; policy = EMA (0.999 per step) of the weights. The $\sigma(s)$ head sits on
detached features; PPO replaces it by a constant per dimension.

\textbf{2. DAgger.} With probability $p_{\text{lab}}$ after a (noisy) policy episode: replay its actions up to
$t_{\text{sw}}\sim U[0,t_{\text{fail}}-0.2\,\text{s}]$ ($t_{\text{fail}}$: first collision, else grasp, else end); (C)
then plans from the robot's moving $(q,v)$, simulation paused, $\le10$\,s wall time per plan. If (C) succeeds, its
actions from $t_{\text{sw}}$ on are labels: 3.3M from 40k BC rollouts before PPO, 5.0M in all.

\textbf{3. PPO + imitation.} One actor loss, $L=w\,L_{\text{PPO}}+\lambda\,(L_\mu+L_\sigma)-\sum_i c_i\log\sigma_i$, per
minibatch of 4096 PPO rows and 4096 imitation rows (half (C) demos, half DAgger labels):
\[
  L_{\text{PPO}}=-\hat{\mathbb E}\big[\omega\min\!\big(r\hat A,\operatorname{clip}(r,0.8,1.2)\hat A\big)\big],\ \
  L_\mu=\hat{\mathbb E}\tfrac{\sum_i(\tanh\mu_i(s')-a^*_i)^2}{2\bar\sigma^2},\ \
  L_\sigma=\hat{\mathbb E}\textstyle\sum_i\Big[\tfrac{\min(\rho_i^2,9\sigma_i^2)}{2\sigma_i^2}+\log\sigma_i\Big].
\]
\begin{itemize}
  \item Decoupled PPO (Hilton et al.\ 2021): $r=\pi_\theta/\pi_{\text{prox}}$,
    $\omega=\min(\pi_{\text{prox}}/\pi_{\text{beh}},2)$; $\pi_{\text{prox}}$ is the policy at the update's start,
    $\pi_{\text{beh}}$ the older one the workers acted with. $\hat A$: GAE ($\gamma=0.99$, 0.95),
    normalised. 4 epochs per batch of 32{,}768 steps, stopped once the KL exceeds 0.03.
  \item $L_\mu$: the expert's likelihood at the policy's noise, $\bar\sigma^2=\sg(\text{mean joint }\sigma_i^2)$.
    $L_\sigma$: the expert residual $\rho_i=\operatorname{atanh}a^*_i-\sg\mu_i$ (gripper $a^*_8-\tanh\sg\mu_8$) sets
    $\sigma$; gripper $\sigma_8\ge0.018$.
  \item Critic: own 3$\times$512 LayerNorm MLP, $(V_\phi(s)-\hat R)^2$ on the GAE returns of the last 2 batches,
    pretrained on the 40k BC rollouts.
  \item Adam: actor $5\cdot10^{-5}$ (halved from batch 747), $\log\sigma$ $10^{-3}$, $V$ $3\cdot10^{-4}$. Exported
    policy: EMA (0.999). $\sim$400 CPU workers (simulation, (C) takeovers) feed one GPU learner asynchronously.
\end{itemize}
\textbf{Schedule} (batch $b$):
\begin{itemize}
  \item BC + DAgger only ($w=0$, $\lambda=0.04$, $p_{\text{lab}}=0.25$) until $V$'s explained variance on new batches
    is $\ge0.5$ three times in a row, after at least 100 batches: after $b=100$.
  \item Ramp over 300 batches, $x:0\to1$: $w=x$, $\lambda=0.04(1-x)$, $p_{\text{lab}}=0.25(1-x)$, $c_i:0.02\to0.002$
    (gripper $0.01\to0.001$).
  \item Final anneal, $b=1100\to1800$: $c_i$ and the actor and $\log\sigma$ learning rates linearly to 0.
  \item Picked: $b=1300$ (42.7M steps, 3.9\,h), the best success on 500 fresh training-mix scenes (Train-500).
\end{itemize}
\textbf{Rewards:} $r_t=0.937\,\Delta\Phi_p+\Delta\Phi_c+\Delta\Phi_e+r_{\text{close}}+r_{\text{end}}$,
$\Delta\Phi=\gamma\Phi(s_{t+1})-\Phi(s_t)$, $h(d,m)=\max(0,m-d)/m$.
\begin{itemize}
  \item Progress $\Phi_p=-(\lVert p_G-p_{\text{grasp}}\rVert+0.1\,\theta_{\text{err}})$, 0 once in hand
    ($0.937=1/$mean start distance). Clearance $\Phi_c=-0.5\,h(d_{\text{obst}},4\,\text{cm})$; cart and floor
    $\Phi_e=-0.25\max\!\big(h(d_{\text{arm}},2\,\text{cm}),h(d_{\text{hand}},5\,\text{mm})\big)$.
  \item Close: the first close within 1\,cm and 10$^\circ$ of the grasp pose pays $0.5\,e^{-|v_G-v_{\text{obj}}|/0.05}$,
    a close outside them $-0.1$. End: success $+1$, collision $-1$, drop $-1$, timeout 0; a failure after success pays $-2$.
\end{itemize}
